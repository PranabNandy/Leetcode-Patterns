1. sizeof(void) ---> Error except ( GNU C == 1)

```c++
int foo() { }
int main(){
  int *p = new int(100);
  void *v=p; // ok

  cout<<*v<<endl; // error
  cout<<*(char *)v<<endl; // ok

  v= foo();  // error

  const int *p1=new int(100);
  volatile int *p2= new int(101);

  void *v=p1; // error
  void *v=p2; // error

  return 0;
}
```
